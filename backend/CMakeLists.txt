cmake_minimum_required(VERSION 3.16)
project(hearthstone-hud LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG 3.3.10)
FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG v1.90.9)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.11.3)
FetchContent_Declare(stb GIT_REPOSITORY https://github.com/nothings/stb.git GIT_TAG f0569113c93ad095470c54bf34a17b36646bbbb5)
FetchContent_MakeAvailable(glfw imgui json stb)
find_package(Threads REQUIRED)
find_package(OpenGL REQUIRED)
find_package(X11 REQUIRED)
add_library(imgui_lib STATIC ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp)
target_include_directories(imgui_lib PUBLIC ${imgui_SOURCE_DIR})
target_link_libraries(imgui_lib PUBLIC glfw OpenGL::GL ${CMAKE_DL_LIBS})
add_executable(hearthstone-hud-renderer main.cpp)
target_include_directories(hearthstone-hud-renderer PRIVATE ${stb_SOURCE_DIR} ${X11_INCLUDE_DIR})
target_link_libraries(hearthstone-hud-renderer PRIVATE Threads::Threads imgui_lib nlohmann_json::nlohmann_json ${X11_LIBRARIES} ${X11_Xext_LIB})
target_link_options(hearthstone-hud-renderer PRIVATE -static-libstdc++ -static-libgcc)
add_executable(hud-state-tests state_test.cpp)
target_link_libraries(hud-state-tests PRIVATE nlohmann_json::nlohmann_json)
